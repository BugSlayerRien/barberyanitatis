  Future<void> _releaseScannerResources() async {
    if (kIsWeb) {
      await _webCamera.stop();
      if (mounted) {
        setState(() {
          _webCameraReady = false;
        });
      }
      return;
    }

    final camera = _camera;
    if (camera == null) {
      return;
    }

    try {
      if (camera.value.isStreamingImages) {
        await camera.stopImageStream();
      }
    } catch (_) {}

    await camera.dispose();
    if (mounted) {
      setState(() {
        _camera = null;
      });
    } else {
      _camera = null;
    }
  }

  bool _shouldAutoFetchRecommendations() {
    if (_normalizedFaceRect == null || !_faceSessionReady) {
      return false;
    }
    if (_loadingRecommendations || _autoProcessingStarted) {
      return false;
    }
    final lastFetch = _lastRecommendationFetchAt;
    if (lastFetch == null) {
      return true;
    }
    return DateTime.now().difference(lastFetch) >= _recommendationCooldown;
  }

  void _triggerAutoFetch({required bool isWeb}) {
    _autoProcessingStarted = true;
    if (isWeb) {
      _fetchRecommendationsWeb(showToastOnError: false, autoTriggered: true);
    } else {
      _fetchRecommendations(autoTriggered: true);
    }
  }

  String _inferFaceShape(double w, double h) {
    if (h <= 0) return 'Oval';
    final ratio = w / h;
    if (ratio > 0.95) return 'Round';
    if (ratio < 0.70) return 'Oblong';
    if (ratio > 0.84) return 'Square';
    return 'Oval';
  }

  Future<void> _fetchRecommendations({bool autoTriggered = false}) async {
    if (autoTriggered) {
      _autoProcessingStarted = true;
    }

    if (kIsWeb) {
      await _fetchRecommendationsWeb(autoTriggered: autoTriggered);
      return;
    }

    if (_normalizedFaceRect == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Please center your face first.')),
      );
      return;
    }

    if (_deterministicTestingMode) {
      _applyDeterministicRecommendation();
      return;
    }

    setState(() => _loadingRecommendations = true);
    try {
      final response = await http.post(
        Uri.parse('${ApiConfig.baseUrl}/ai_scan.php'),
        headers: {'Content-Type': 'application/json'},
        body: jsonEncode({
          'action': 'rankByFeatures',
          'faceShape': _faceShape,
          'hairTexture': _hairTexture,
          'hairDensity': UserSession.lastScan?['hairDensity']?.toString(),
        }),
      );

      final data = response.body.isNotEmpty
          ? Map<String, dynamic>.from(jsonDecode(response.body) as Map)
          : <String, dynamic>{};

      if (response.statusCode != 200) {
        throw Exception(
            data['error']?.toString() ?? 'Failed to fetch recommendations');
      }

      final variants = (data['recommendations'] as List<dynamic>? ??
              data['variants'] as List<dynamic>? ??
              [])
          .map((e) => Map<String, dynamic>.from(e as Map))
          .toList();

      if (variants.isEmpty) {
        final fallback = await _loadLibraryFallback();
        final rankedFallback = _rankRecommendations(
            fallback.isEmpty ? _hardcodedFallback() : fallback);
        if (!mounted) return;
        setState(() {
          _recommendations
            ..clear()
            ..addAll(rankedFallback.take(3));
          _selectedIndex = 0;
          _markOverlayReadyAfterProcessing();
          _status = 'Suggestions are ready. Pick one to preview.';
        });
      } else {
        final ranked = _rankRecommendations(variants);
        if (!mounted) return;
        setState(() {
          _recommendations
            ..clear()
            ..addAll(ranked.take(3));
          _selectedIndex = 0;
          _markOverlayReadyAfterProcessing();
          _status = 'Suggestions are ready. Pick one to preview.';
        });
      }

      UserSession.lastScan = {
        'faceShape': _faceShape,
        'hairTexture': _hairTexture,
        'headYaw': _headYaw,
      };
      UserSession.lastRecommendations = _recommendations;

      UserSession.activeScanSessionId = null;

      await AiFunnelService.trackScan(
        scanSessionId: UserSession.ensureScanSessionId(),
        userId: UserSession.userId,
        faceShape: _faceShape,
        hairTexture: _hairTexture,
        hairDensity: null,
        hairLength: null,
        recommendationCount: _recommendations.length,
      );

      _lastRecommendationFetchAt = DateTime.now();
    } catch (e) {
      if (autoTriggered) {
        _autoProcessingStarted = false;
      }
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Failed to load recommendations: $e')),
      );
    } finally {
      if (mounted) {
        setState(() => _loadingRecommendations = false);
      }
    }
  }

  Future<void> _fetchRecommendationsWeb(
      {bool showToastOnError = true, bool autoTriggered = false}) async {
    if (autoTriggered) {
      _autoProcessingStarted = true;
    }

    if (_normalizedFaceRect == null) {
      if (showToastOnError && mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
              content: Text(
                  'Center your face first so we can continue.')),
        );
      }
      return;
    }

    if (!_webCameraReady) {
      if (showToastOnError) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
              content:
                  Text('Camera is not ready yet. Please allow camera access.')),
        );
      }
      return;
    }

    if (_deterministicTestingMode) {
      _applyDeterministicRecommendation();
      return;
    }

    setState(() => _loadingRecommendations = true);
    try {
      if (mounted) {
        setState(() {
          _status = 'Updating haircut suggestions...';
        });
      }

      final response = await http.post(
        Uri.parse('${ApiConfig.baseUrl}/ai_scan.php'),
        headers: {'Content-Type': 'application/json'},
        body: jsonEncode({
          'action': 'rankByFeatures',
          'faceShape': _faceShape,
          'hairTexture': _hairTexture,
          'hairDensity': UserSession.lastScan?['hairDensity']?.toString(),
        }),
      );

      final data = response.body.isNotEmpty
          ? Map<String, dynamic>.from(jsonDecode(response.body) as Map)
          : <String, dynamic>{};

      if (response.statusCode != 200) {
        throw Exception(data['error']?.toString() ??
            'Failed to fetch live recommendations');
      }

      var recommendations = (data['recommendations'] as List<dynamic>? ??
              data['variants'] as List<dynamic>? ??
              [])
          .map((e) => Map<String, dynamic>.from(e as Map))
          .toList();

      if (recommendations.isEmpty) {
        recommendations = await _loadLibraryFallback();
      }

      if (recommendations.isEmpty) {
        recommendations = _hardcodedFallback();
      }

      if (recommendations.isEmpty) {
        throw Exception('No available hairstyle variants yet');
      }

      final ranked = _rankRecommendations(recommendations);

      if (!mounted) return;
      setState(() {
        _recommendations
          ..clear()
          ..addAll(ranked.take(3));
        _selectedIndex = 0;
        _markOverlayReadyAfterProcessing();
        _status =
          'Suggestions are ready. Hair preview now uses your selected style.';
      });

      UserSession.lastScan = {
        'faceShape': _faceShape,
        'hairTexture': _hairTexture,
      };
      UserSession.lastRecommendations = recommendations;

      UserSession.activeScanSessionId = null;

      await AiFunnelService.trackScan(
        scanSessionId: UserSession.ensureScanSessionId(),
        userId: UserSession.userId,
        faceShape: _faceShape,
        hairTexture: _hairTexture,
        hairDensity: null,
        hairLength: null,
        recommendationCount: _recommendations.length,
      );

      _lastRecommendationFetchAt = DateTime.now();
    } catch (e) {
      if (autoTriggered) {
        _autoProcessingStarted = false;
      }
      if (!mounted) return;
      setState(() {
        _status = 'Could not update suggestions. Please try again.';
      });
      if (showToastOnError) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Live sync failed: $e')),
        );
      }
    } finally {
      if (mounted) {
        setState(() => _loadingRecommendations = false);
      }
    }
  }

  Future<List<Map<String, dynamic>>> _loadLibraryFallback() async {
    final response = await http.post(
      Uri.parse('${ApiConfig.baseUrl}/haircuts.php'),
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode({'action': 'listLibrary'}),
    );

    if (response.statusCode != 200 || response.body.isEmpty) {
      return _hardcodedFallback();
    }

    final data = jsonDecode(response.body) as Map<String, dynamic>;
    final styles = (data['hairstyles'] as List<dynamic>? ?? [])
        .map((e) => Map<String, dynamic>.from(e as Map))
        .toList();
    if (styles.isEmpty) {
      return _hardcodedFallback();
    }
    return styles.take(8).toList();
  }

  List<Map<String, dynamic>> _hardcodedFallback() {
    return _hardcodedCatalog.map((e) => Map<String, dynamic>.from(e)).toList();
  }

  Widget _buildHairOnlyOverlayModel(
      {required OverlayCalibration calibration,
      required Map<String, dynamic>? recommendation}) {
    if (_shouldUse3DModel(recommendation)) {
      return _buildHair3DOverlayModel(calibration: calibration);
    }

    final yaw = (_headYaw ?? 0).clamp(-60.0, 60.0);
    final yawAbsNorm = (yaw.abs() / 34.0).clamp(0.0, 1.0);
    final frontOpacity = (1.0 - (yawAbsNorm * 0.58)).clamp(0.42, 1.0);
    final sideOpacity = (0.16 + (yawAbsNorm * 0.84)).clamp(0.0, 1.0);

    final frontImage = _sampleImageOverlay.front;
    final sideImage = yaw >= 0 ? _sampleImageOverlay.left : _sampleImageOverlay.right;
    final sideAlignment = yaw >= 0 ? Alignment.centerLeft : Alignment.centerRight;

    return ClipRect(
      child: Align(
        alignment: Alignment.topCenter,
        heightFactor: calibration.hairClipFactor,
        child: SizedBox.expand(
          child: Transform.scale(
            scale: calibration.visualScale,
            alignment: Alignment.topCenter,
            child: Stack(
              fit: StackFit.expand,
              children: [
                Opacity(
                  opacity: frontOpacity,
                  child: Image.asset(
                    frontImage,
                    fit: BoxFit.contain,
                    alignment: Alignment.topCenter,
                    filterQuality: FilterQuality.medium,
                    errorBuilder: (context, error, stackTrace) {
                      return const ColoredBox(color: Colors.transparent);
                    },
                  ),
                ),
                Opacity(
                  opacity: sideOpacity,
                  child: Image.asset(
                    sideImage,
                    fit: BoxFit.contain,
                    alignment: Alignment.topCenter,
                    filterQuality: FilterQuality.medium,
                    errorBuilder: (context, error, stackTrace) {
                      return const ColoredBox(color: Colors.transparent);
                    },
                  ),
                ),
                if (yawAbsNorm > 0.08)
                  Align(
                    alignment: sideAlignment,
                    child: ClipRect(
                      child: Align(
                        alignment: sideAlignment,
                        widthFactor: 0.46,
                        child: Opacity(
                          opacity: (yawAbsNorm * 0.48).clamp(0.0, 0.48),
                          child: Transform.scale(
                            scaleX: 1.14,
                            scaleY: 1.03,
                            alignment: sideAlignment,
                            child: Image.asset(
                              sideImage,
                              fit: BoxFit.cover,
                              alignment: sideAlignment,
                              filterQuality: FilterQuality.medium,
                              errorBuilder: (context, error, stackTrace) {
                                return const ColoredBox(
                                    color: Colors.transparent);
                              },
                            ),
                          ),
                        ),
                      ),
                    ),
                  ),
              ],
            ),
          ),
        ),
      ),
    );
  }

  bool _shouldUse3DModel(Map<String, dynamic>? recommendation) {
    if (!_enable3DHairOverlay) {
      return false;
    }
    if (recommendation == null) {
      return false;
    }
    final styleName = (recommendation['style_name']?.toString() ??
            recommendation['styleName']?.toString() ??
            '')
        .toLowerCase();

    // Initial 3D integration targets the provided blowout taper sample model.
    return styleName.contains('blowout taper');
  }

  Widget _buildHair3DOverlayModel({required OverlayCalibration calibration}) {
    final yaw = (_headYaw ?? 0).clamp(-60.0, 60.0);
    final pitch = ((_headRoll ?? 0) * 0.2).clamp(-15.0, 15.0);
    final orbit = '${yaw.toStringAsFixed(1)}deg ${pitch.toStringAsFixed(1)}deg 86%';

    return ClipRect(
      child: Align(
        alignment: Alignment.topCenter,
        heightFactor: calibration.hairClipFactor,
        child: SizedBox.expand(
          child: Transform.scale(
            scale: calibration.visualScale,
            alignment: Alignment.topCenter,

              child: ModelViewer(
                src: _sampleHairModel3d,
                alt: '3D hair overlay',
                cameraControls: false,
                disableZoom: true,
                autoRotate: false,
                cameraOrbit: orbit,
                interactionPrompt: InteractionPrompt.none,
                backgroundColor: const Color.fromARGB(0, 0, 0, 0),
              ),
            ),
          ),
        ),
      ),
    );
  }

  void _applyDeterministicRecommendation() {
    final fallback = _hardcodedFallback();
    if (fallback.isEmpty || !mounted) {
      return;
    }

    final defaultStyle = fallback.first;
    setState(() {
      _loadingRecommendations = false;
      _recommendations
        ..clear()
        ..add(defaultStyle);
      _selectedIndex = 0;
      _status = 'Deterministic mode: default haircut applied for testing.';
    });

    UserSession.lastScan = {
      'faceShape': _faceShape,
      'hairTexture': _hairTexture,
      'headYaw': _headYaw,
    };
    UserSession.lastRecommendations = [defaultStyle];
  }

  void _markOverlayReadyAfterProcessing() {
    final now = DateTime.now();
    final bestFace = _normalizedFaceRect ?? _smoothedFaceRect;
    if (bestFace == null) {
      return;
    }
    _lastStableFaceRect = bestFace;
    _overlayHoldUntil = now.add(_overlayHoldAfterProcessDuration);
  }

  Widget _buildScannerCard() {
    final recommendation =
        _recommendations.isNotEmpty ? _recommendations[_selectedIndex] : null;
    final overlayFace = _effectiveOverlayFaceRect();
    final overlayOpacity = _currentOverlayOpacity();
    final scannerHeight =
      (MediaQuery.of(context).size.height * 0.58).clamp(340.0, 620.0);

    Widget? overlayLayer(BoxConstraints constraints) {
      if (overlayFace == null || recommendation == null || overlayOpacity <= 0.01) {
        return null;
      }

      final calibration = _overlayCalibration(recommendation);
      final box = _computeAttachmentBox(
        face: overlayFace,
        constraints: constraints,
        calibration: calibration,
      );
      return Positioned(
        left: box.left,
        top: box.top,
        width: box.width,
        height: box.height,
        child: IgnorePointer(
          child: Opacity(
            opacity: overlayOpacity,
            child: Transform(
              alignment: Alignment.center,
              transform: Matrix4.identity()
                ..rotateZ((_headRoll ?? 0) * math.pi / 180)
                ..scale(box.yawScale, 1.0),
              child: _buildHairOnlyOverlayModel(
                calibration: calibration,
                recommendation: recommendation,
              ),
            ),
          ),
        ),
      );
    }

    Widget? hairBlurLayer(BoxConstraints constraints) {
      if (overlayFace == null || recommendation == null || overlayOpacity <= 0.01) {
        return null;
      }

      // BackdropFilter does not reliably blur HtmlElementView camera layers on web.
      // Use the cover layer below for consistent real-hair suppression on web.
      if (kIsWeb) {
        return null;
      }

      final calibration = _overlayCalibration(recommendation);
      final box = _computeAttachmentBox(
        face: overlayFace,
        constraints: constraints,
        calibration: calibration,
      );

      final blurLeft = (box.left + (box.width * 0.10))
          .clamp(0.0, constraints.maxWidth)
          .toDouble();
      final blurTop = (box.top + (box.height * 0.12))
          .clamp(0.0, constraints.maxHeight)
          .toDouble();
      final blurWidth = (box.width * 0.80)
          .clamp(0.0, constraints.maxWidth - blurLeft)
          .toDouble();
        final blurHeight = (box.height * 0.44)
          .clamp(0.0, constraints.maxHeight - blurTop)
          .toDouble();

      if (blurWidth <= 1 || blurHeight <= 1) {
        return null;
      }

      return Positioned(
        left: blurLeft,
        top: blurTop,
        width: blurWidth,
        height: blurHeight,
        child: IgnorePointer(
          child: ClipRRect(
            borderRadius: BorderRadius.circular(999),
            child: BackdropFilter(
              filter: ui.ImageFilter.blur(
                  sigmaX: _hairBlurSigma, sigmaY: _hairBlurSigma),
              child: Container(color: Colors.transparent),
            ),
          ),
        ),
      );
    }

    Widget? hairCoverLayer(BoxConstraints constraints) {
      if (overlayFace == null || recommendation == null || overlayOpacity <= 0.01) {
        return null;
      }

      // Browser camera layers can render differently across window/fullscreen.
      // Keep web path clean to avoid misplaced dark masks.
      if (kIsWeb) {
        return null;
      }

      final calibration = _overlayCalibration(recommendation);
      final box = _computeAttachmentBox(
        face: overlayFace,
        constraints: constraints,
        calibration: calibration,
      );

      final coverLeft = (box.left + (box.width * 0.08))
          .clamp(0.0, constraints.maxWidth)
          .toDouble();
      final coverTop = (box.top + (box.height * 0.10))
          .clamp(0.0, constraints.maxHeight)
          .toDouble();
      final coverWidth = (box.width * 0.84)
          .clamp(0.0, constraints.maxWidth - coverLeft)
          .toDouble();
        final coverHeight = (box.height * 0.46)
          .clamp(0.0, constraints.maxHeight - coverTop)
          .toDouble();

      if (coverWidth <= 1 || coverHeight <= 1) {
        return null;
      }

      return Positioned(
        left: coverLeft,
        top: coverTop,
        width: coverWidth,
        height: coverHeight,
        child: IgnorePointer(
          child: ClipRRect(
            borderRadius: BorderRadius.circular(999),
            child: Stack(
              fit: StackFit.expand,
              children: [
                Container(
                  decoration: BoxDecoration(
                    gradient: LinearGradient(
                      begin: Alignment.topCenter,
                      end: Alignment.bottomCenter,
                      colors: [
                        Colors.black.withOpacity(_hairCoverTopOpacity),
                        Colors.black.withOpacity(_hairCoverMidOpacity),
                        Colors.black.withOpacity(_hairCoverBottomOpacity),
                      ],
                    ),
                  ),
                ),
                Container(
                  decoration: BoxDecoration(
                    gradient: RadialGradient(
                      center: const Alignment(0, -0.55),
                      radius: 1.05,
                      colors: [
                        Colors.transparent,
                        Colors.black.withOpacity(_hairCoverRadialOpacity),
                      ],
                    ),
                  ),
                ),
              ],
            ),
          ),
        ),
      );
    }

          .toDouble();

          .clamp(0.0, constraints.maxHeight - featherTop)

        return null;

  List<Map<String, dynamic>> _rankRecommendations(
      List<Map<String, dynamic>> input) {
    final ranked = List<Map<String, dynamic>>.from(input);

    double score(Map<String, dynamic> item) {
      final confidence = double.tryParse(
              '${item['confidence_score'] ?? item['confidenceScore'] ?? 0}') ??
          0;
      final styleName = (item['style_name']?.toString() ??
              item['styleName']?.toString() ??
              '')
          .toLowerCase();
      final face = (item['face_shape']?.toString() ??
              item['faceShape']?.toString() ??
              '')
          .toLowerCase();
      final texture = (item['hair_texture']?.toString() ??
              item['hairTexture']?.toString() ??
              '')
          .toLowerCase();

      var s = confidence;
      if (face.isNotEmpty && face == _faceShape.toLowerCase()) {
        s += 0.20;
      }
      if (texture.isNotEmpty && texture == _hairTexture.toLowerCase()) {
        s += 0.25;
      }

      final likelyCoilyStyle = styleName.contains('afro') ||
          styleName.contains('dread') ||
          styleName.contains('twist');
      if (_hairTexture.toLowerCase() != 'coily' && likelyCoilyStyle) {
        s -= 0.55;
      }

      return s;
    }

    ranked.sort((a, b) => score(b).compareTo(score(a)));
    return ranked;
  }

  Rect _smoothRect(Rect? previous, Rect next, double alpha) {
    if (previous == null) {
      return next;
    }
    return Rect.fromLTWH(
      _smoothValue(previous.left, next.left, alpha),
      _smoothValue(previous.top, next.top, alpha),
      _smoothValue(previous.width, next.width, alpha),
      _smoothValue(previous.height, next.height, alpha),
    );
  }

  double _smoothValue(double? previous, double? next, double alpha) {
    final target = next ?? 0;
    if (previous == null) {
      return target;
    }
    final value = previous + ((target - previous) * alpha);
    return value;
  }

  Rect _stabilizeIncomingRect(Rect? previous, Rect next) {
    if (previous == null) {
      return next;
    }

    final prevCenter = previous.center;
    final nextCenter = next.center;
    final centerJump = (nextCenter - prevCenter).distance;
    final areaJump = ((_rectArea(next) - _rectArea(previous)).abs() /
            (_rectArea(previous).clamp(0.001, 1.0)))
        .clamp(0.0, 2.0);

    if (centerJump > _trackingJumpThreshold || areaJump > 0.40) {
      return Rect.lerp(previous, next, 0.22) ?? next;
    }
    return next;
  }

  bool _isLikelyStableFace(Rect rect) {
    final area = _rectArea(rect);
    if (area < _minFaceAreaForOverlay) {
      return false;
    }
    const maxArea =
        kIsWeb ? _maxFaceAreaForOverlayWeb : _maxFaceAreaForOverlayMobile;
    if (area > maxArea) {
      return false;
    }
    if (rect.left <= 0.01 ||
        rect.top <= 0.01 ||
        rect.right >= 0.99 ||
        rect.bottom >= 0.99) {
      return false;
    }
    return true;
  }

  double _rectArea(Rect rect) => (rect.width * rect.height).clamp(0.0, 1.0);

      if (_isLikelyStableFace(current)) {

        _overlayHoldUntil = DateTime.now().add(_overlayHoldDuration);
        return current;

      final fallback = _lastStableFaceRect;
      if (holdUntil != null && fallback != null && DateTime.now().isBefore(holdUntil)) {
        return fallback;

      return current;

    final fallback = _lastStableFaceRect;
    if (holdUntil == null || fallback == null) {

      return null;
    }
    return fallback;

    if (remainingMs <= 0) {
      return 0;

        (remainingMs / _overlayHoldDuration.inMilliseconds).clamp(0.0, 1.0);

  OverlayCalibration _adaptCalibrationToFace(OverlayCalibration base) {
    final face = _effectiveOverlayFaceRect() ?? _smoothedFaceRect ?? _normalizedFaceRect;
    if (face == null) {
      return base;
    }

    final area = (face.width * face.height).clamp(0.03, 0.40);
    final aspect = (face.width / face.height).clamp(0.58, 1.15);

    final slimFaceNorm = ((0.88 - aspect) / 0.30).clamp(0.0, 1.0);
    final wideFaceNorm = ((aspect - 0.84) / 0.24).clamp(0.0, 1.0);
    final nearNorm = ((area - 0.15) / 0.18).clamp(0.0, 1.0);
    final farNorm = ((0.15 - area) / 0.12).clamp(0.0, 1.0);
    final eyeDistance =
      (_eyeDistanceNorm ?? (face.width * 0.42)).clamp(0.05, 0.70);
    final expectedEye = (face.width * 0.42).clamp(0.05, 0.70);
    final eyeScaleNorm =
      ((expectedEye - eyeDistance).abs() / expectedEye).clamp(0.0, 1.0);
    final eyeIsWider = (eyeDistance > expectedEye) ? 1.0 : 0.0;

    final widthScale = (base.widthScale *
            (1.0 + (slimFaceNorm * 0.18) - (wideFaceNorm * 0.08)) *
        (1.0 + (farNorm * 0.10) - (nearNorm * 0.06)) *
        (1.0 - (eyeScaleNorm * 0.06 * eyeIsWider)))
        .clamp(0.96, 1.58)
        .toDouble();

    final heightScale = (base.heightScale *
            (1.0 + (wideFaceNorm * 0.16) + (slimFaceNorm * 0.05)) *
        (1.0 + (nearNorm * 0.08) + (farNorm * 0.04)) *
        (1.0 + (eyeScaleNorm * 0.05 * (1.0 - eyeIsWider))))
        .clamp(1.04, 1.78)
        .toDouble();

    final yOffset = (base.yOffset +
            (nearNorm * 0.04) -
            (farNorm * 0.03) -
            (slimFaceNorm * 0.02))
      .clamp(-0.30, -0.08)
        .toDouble();

    final clipFactor =
        (base.hairClipFactor + (wideFaceNorm * 0.03) - (slimFaceNorm * 0.02))
            .clamp(0.80, 0.92)
            .toDouble();

    final visualScale =
        (base.visualScale + (farNorm * 0.06) - (nearNorm * 0.04))
        .clamp(1.00, 1.22)
            .toDouble();

    return OverlayCalibration(
      widthScale: widthScale,
      heightScale: heightScale,
      xOffset: base.xOffset,
      yOffset: yOffset,
      hairClipFactor: clipFactor,
      visualScale: visualScale,
      yawShiftFactor: base.yawShiftFactor,
    );
  }

class MatrixAttachmentEngine extends AttachmentEngine {
  final AttachmentEngine fallback;

  const MatrixAttachmentEngine({this.fallback = const HeuristicAttachmentEngine()});

  @override
  OverlayBox compute(AttachmentInput input) {
    final matrix = input.faceTransformMatrix;
    if (matrix == null || matrix.length < 16) {
      return fallback.compute(input);
    }

    final scaleX = math.sqrt((matrix[0] * matrix[0]) + (matrix[1] * matrix[1]));
    final scaleY = math.sqrt((matrix[4] * matrix[4]) + (matrix[5] * matrix[5]));
    final matrixScale = ((scaleX + scaleY) * 0.5).clamp(0.55, 1.55);
    final inferredRoll = math.atan2(matrix[1], matrix[0]) * (180 / math.pi);
    final tx = matrix[12].clamp(-1.5, 1.5);
    final ty = matrix[13].clamp(-1.5, 1.5);

    final adjustedCalibration = OverlayCalibration(
      widthScale: (input.calibration.widthScale * (1.0 + ((matrixScale - 1.0) * 0.14)))
          .clamp(0.90, 1.75)
          .toDouble(),
      heightScale:
          (input.calibration.heightScale * (1.0 + ((matrixScale - 1.0) * 0.11)))
              .clamp(0.98, 1.95)
              .toDouble(),
      xOffset: (input.calibration.xOffset + (tx * 0.035)).clamp(-0.16, 0.16).toDouble(),
      yOffset: (input.calibration.yOffset + (ty * 0.030)).clamp(-0.36, 0.10).toDouble(),
      hairClipFactor: input.calibration.hairClipFactor,
      visualScale: input.calibration.visualScale,
      yawShiftFactor: input.calibration.yawShiftFactor,
    );

    return computeOverlayBox(
      face: input.face,
      constraints: input.constraints,
      calibration: adjustedCalibration,
      isWeb: input.isWeb,
      headYaw: input.headYaw,
      headRoll: input.headRoll ?? inferredRoll,
      foreheadAnchorY: input.foreheadAnchorY,
      eyeMidX: input.eyeMidX,
      eyeDistanceNorm: input.eyeDistanceNorm,
      videoAspect: input.videoAspect,
    );
  }
}

  var mappedEyeDistanceNorm = eyeDistanceNorm;
  var mappedForeheadAnchorY = foreheadAnchorY;

  // Web tracker coordinates are normalized to the raw video frame.
  // Convert them into container-normalized space when the video is letterboxed.
  if (isWeb && videoAspect != null && videoAspect > 0) {
    final cw = constraints.maxWidth;
    final ch = constraints.maxHeight;
    final containerAspect = cw / ch;

    var displayW = cw;
    var displayH = ch;
    var offsetX = 0.0;
    var offsetY = 0.0;

    if (containerAspect > videoAspect) {
      displayH = ch;
      displayW = displayH * videoAspect;
      offsetX = (cw - displayW) * 0.5;
    } else {
      displayW = cw;
      displayH = displayW / videoAspect;
      offsetY = (ch - displayH) * 0.5;
    }

    final mappedLeft = (offsetX + (face.left * displayW)) / cw;
    final mappedTop = (offsetY + (face.top * displayH)) / ch;
    final mappedWidth = (face.width * displayW) / cw;
    final mappedHeight = (face.height * displayH) / ch;
    mappedFace = Rect.fromLTWH(mappedLeft, mappedTop, mappedWidth, mappedHeight);

    if (eyeMidX != null) {
      mappedEyeMidX = ((offsetX + (eyeMidX * displayW)) / cw).clamp(0.0, 1.0);
    }
    if (eyeDistanceNorm != null) {
      mappedEyeDistanceNorm = ((eyeDistanceNorm * displayW) / cw).clamp(0.02, 0.90);
    }
    if (foreheadAnchorY != null) {
      mappedForeheadAnchorY =
          ((offsetY + (foreheadAnchorY * displayH)) / ch).clamp(0.0, 1.0);
    }
  }

  final faceWidth = mappedFace.width * constraints.maxWidth;
  final faceHeight = mappedFace.height * constraints.maxHeight;
  final faceAspect = (mappedFace.width / mappedFace.height).clamp(0.55, 1.35);

  final faceAreaNorm = (mappedFace.width * mappedFace.height).clamp(0.03, 0.45);
  final targetArea =
      isWeb ? kOverlayTargetFaceAreaWeb : kOverlayTargetFaceAreaMobile;

  final distanceFactor = (targetArea / faceAreaNorm).clamp(0.72, 1.28);
  final distanceVerticalFactor =
      1.0 + ((faceAreaNorm - targetArea) * 1.2).clamp(-0.12, 0.18);

  final slimFaceNorm = ((0.90 - faceAspect) / 0.40).clamp(0.0, 1.0);
  final wideFaceNorm = ((faceAspect - 0.82) / 0.35).clamp(0.0, 1.0);
  final nearCameraNorm =
    ((faceAreaNorm - targetArea) / targetArea).clamp(0.0, 1.0);
  final farCameraNorm =
    ((targetArea - faceAreaNorm) / targetArea).clamp(0.0, 1.0);

  final adaptiveWidthBoost =
    (1.0 + (slimFaceNorm * 0.22) - (wideFaceNorm * 0.08));
  final adaptiveHeightBoost =
    (1.0 + (wideFaceNorm * 0.16) + (slimFaceNorm * 0.04));
  final distanceWidthComp =
    (1.0 + (farCameraNorm * 0.12) - (nearCameraNorm * 0.06));
  final distanceHeightComp =
    (1.0 + (nearCameraNorm * 0.10) + (farCameraNorm * 0.06));

  final yawDeg = headYaw ?? 0;
  final yawAbsNorm = (yawDeg.abs() / 45.0).clamp(0.0, 1.0);
  final yawWidthFactor = (1.0 - (yawAbsNorm * 0.35)).clamp(0.55, 1.0);
  final yawHeightFactor = (1.0 - (yawAbsNorm * 0.12)).clamp(0.85, 1.0);
  final yawHorizontalShift = (yawDeg / 55.0).clamp(-0.12, 0.12);

    final eyeDistNorm =
      (mappedEyeDistanceNorm ?? (mappedFace.width * 0.42)).clamp(0.05, 0.70);
  final eyeDistPx = eyeDistNorm * constraints.maxWidth;
  final widthFromEyes = eyeDistPx * 2.65;
  final eyeWidthBlend = isWeb ? 0.66 : 0.58;

  final rollDeg = headRoll ?? 0;
  final rollNorm = (rollDeg.abs() / 25.0).clamp(0.0, 1.0);
  final rollTiltFactor = rollDeg.sign;

  final width = faceWidth *
      calibration.widthScale *
      distanceFactor *
      adaptiveWidthBoost *
      distanceWidthComp *
      yawWidthFactor *
      (1.0 - (rollNorm * 0.04));
    final eyeAnchoredWidth =
      (width * (1.0 - eyeWidthBlend)) + (widthFromEyes * eyeWidthBlend);

  final height = faceHeight *
      calibration.heightScale *
      distanceFactor *
      adaptiveHeightBoost *
      distanceHeightComp *
      yawHeightFactor *
      (1.0 + (rollNorm * 0.03));

    final clampedWidth = eyeAnchoredWidth
      .clamp(faceWidth * 0.82, faceWidth * 1.98);
    final clampedHeight = height.clamp(faceHeight * 0.95, faceHeight * 2.05);

    final anchorXNorm =
      (mappedEyeMidX ?? (mappedFace.left + (mappedFace.width * 0.5)))
        .clamp(0.0, 1.0);
  final faceCenterX = anchorXNorm * constraints.maxWidth;
  final xOffsetPx = faceWidth * (calibration.xOffset + yawHorizontalShift);
  final left = (faceCenterX - (clampedWidth * 0.5) + xOffsetPx)
      .clamp(-clampedWidth * 0.15, constraints.maxWidth - (clampedWidth * 0.85));

  double anchorYNorm;
  if (mappedForeheadAnchorY != null) {
    anchorYNorm = mappedForeheadAnchorY.clamp(0.0, 1.0);
  } else {
    anchorYNorm = (mappedFace.top + (mappedFace.height * 0.12)).clamp(0.0, 1.0);
  }

  final yawAnchorShift = -yawAbsNorm * 0.02 * yawDeg.sign;
  final adjustedAnchorYNorm = (anchorYNorm + yawAnchorShift).clamp(0.02, 0.35);
  final anchorY = adjustedAnchorYNorm * constraints.maxHeight;

  final topRaw = anchorY - (clampedHeight * 0.10) +
      (faceHeight * calibration.yOffset * distanceVerticalFactor) +
      (rollTiltFactor * rollNorm * faceHeight * 0.02);

  final top = topRaw
      .clamp(-clampedHeight * 0.35, constraints.maxHeight - (clampedHeight * 0.15));

  return OverlayBox(
    left: left.toDouble(),
    top: top.toDouble(),
    width: clampedWidth.toDouble(),
    height: clampedHeight.toDouble(),
    yawScale: yawWidthFactor,
  );
}

class DetectedFaceFrame {
  final Rect boundingBox;
  final Size imageSize;
  final double? headEulerAngleY;
  final double? headEulerAngleZ;

  const DetectedFaceFrame({
    required this.boundingBox,
    required this.imageSize,
    this.headEulerAngleY,
    this.headEulerAngleZ,
  });
}

class FaceDetectorService {
  late final FaceDetector _faceDetector;

  Future<void> initialize() async {
    _faceDetector = FaceDetector(
      options: FaceDetectorOptions(
        performanceMode: FaceDetectorMode.fast,
        enableLandmarks: true,
        enableContours: false,
        enableTracking: true,
      ),
    );
  }

  Future<DetectedFaceFrame?> processFrame(
    CameraImage image, {
    required int sensorOrientation,
  }) async {
    final inputImage = _toInputImage(image, sensorOrientation);
    if (inputImage == null) {
      return null;
    }

    final faces = await _faceDetector.processImage(inputImage);
    if (faces.isEmpty) {
      return null;
    }

    final face = faces.first;
    return DetectedFaceFrame(
      boundingBox: face.boundingBox,
      imageSize: Size(image.width.toDouble(), image.height.toDouble()),
      headEulerAngleY: face.headEulerAngleY,
      headEulerAngleZ: face.headEulerAngleZ,
    );
  }

  InputImage? _toInputImage(CameraImage image, int sensorOrientation) {
    final format = InputImageFormatValue.fromRawValue(image.format.raw);
    if (format == null) {
      return null;
    }

    final rotation = InputImageRotationValue.fromRawValue(sensorOrientation);
    if (rotation == null) {
      return null;
    }

    final bytes = _concatenatePlanes(image.planes);
    final metadata = InputImageMetadata(
      size: Size(image.width.toDouble(), image.height.toDouble()),
      rotation: rotation,
      format: format,
      bytesPerRow: image.planes.first.bytesPerRow,
    );

    return InputImage.fromBytes(bytes: bytes, metadata: metadata);
  }

  Uint8List _concatenatePlanes(List<Plane> planes) {
    final buffer = BytesBuilder(copy: false);
    for (final plane in planes) {
      buffer.add(plane.bytes);
    }
    return buffer.takeBytes();
  }

  Future<void> dispose() async {
    await _faceDetector.close();
  }
}

  html.VideoElement? _video;
  html.MediaStream? _stream;
  String? _viewType;
  Timer? _trackingTimer;

  final ValueNotifier<WebFaceTrackingState> tracking =
      ValueNotifier<WebFaceTrackingState>(const WebFaceTrackingState());

  bool get isStreaming => _stream != null;

  Future<bool> start() async {
    if (_stream != null) {
      return true;
    }
    final media = html.window.navigator.mediaDevices;
    if (media == null) {
      return false;
    }

    _stream = await media.getUserMedia({
      'video': {
        'facingMode': 'user',
        'width': {'ideal': 1280},
        'height': {'ideal': 720},

      'audio': false,
    });

    _video = html.VideoElement()
      ..autoplay = true
      ..muted = true
      ..style.width = '100%'
      ..style.height = '100%'
      ..srcObject = _stream;
    _video!.setAttribute('playsinline', 'true');

    _viewType = 'webcam-${DateTime.now().microsecondsSinceEpoch}';
    // ignore: undefined_prefixed_name
    ui.platformViewRegistry.registerViewFactory(_viewType!, (int viewId) => _video!);

    await _initFaceTracker();
    _startTrackingLoop();

    return true;
  }

  Future<void> stop() async {
    _trackingTimer?.cancel();
    _trackingTimer = null;
    _video?.pause();
    _video?.srcObject = null;
    _stream?.getTracks().forEach((track) => track.stop());
    _stream = null;
    tracking.value = tracking.value.copyWith(clearFace: true);
  }

  Future<void> _initFaceTracker() async {
    try {
      js.context.callMethod('initTatisFaceTracker');
      tracking.value = tracking.value.copyWith(
        trackerReady: true,
        trackerSupported: true,
      );
    } catch (_) {
      tracking.value = tracking.value.copyWith(
        trackerReady: false,
        trackerSupported: false,
        clearFace: true,
      );
    }
  }

  void _startTrackingLoop() {
    _trackingTimer?.cancel();
    _trackingTimer = Timer.periodic(const Duration(milliseconds: 70), (_) {
      _estimateFace();
    });
  }

  void _estimateFace() {
    final video = _video;
    if (video == null || _stream == null) {
      return;
    }

    try {
      final result = js.context.callMethod('tatisEstimateFace', [video]);
      if (result == null) {
        return;
      }

      if (result is! js.JsObject) {
        return;
      }

      final status = (result['status'] as String?)?.toLowerCase();
      if (status == 'stale') {
        return;
      }
      if (status == 'no_face' || status == 'error') {
        tracking.value = tracking.value.copyWith(clearFace: true);
        return;
      }

      final bbox = result['bbox'];
      if (bbox == null) {
        tracking.value = tracking.value.copyWith(clearFace: true);
        return;
      }

      if (bbox is! js.JsObject) {
        tracking.value = tracking.value.copyWith(clearFace: true);
        return;
      }

      final left = ((bbox['left'] as num?) ?? 0).toDouble().clamp(0.0, 1.0);
      final top = ((bbox['top'] as num?) ?? 0).toDouble().clamp(0.0, 1.0);
      final width = ((bbox['width'] as num?) ?? 0).toDouble().clamp(0.0, 1.0);
      final height = ((bbox['height'] as num?) ?? 0).toDouble().clamp(0.0, 1.0);
      final yaw = ((result['yaw'] as num?) ?? 0).toDouble();
      final roll = ((result['roll'] as num?) ?? 0).toDouble();
      final eyeMidY = ((result['eyeMidY'] as num?) ?? (top + (height * 0.36))).toDouble().clamp(0.0, 1.0);
      final eyeMidX = ((result['eyeMidX'] as num?) ?? (left + (width * 0.5)))
          .toDouble()
          .clamp(0.0, 1.0);
      final eyeDistanceNorm =
          ((result['eyeDistanceNorm'] as num?) ?? (width * 0.42))
              .toDouble()
              .clamp(0.04, 0.65);
      final videoAspect = (video.videoHeight <= 0)
          ? (16 / 9)
          : (video.videoWidth / video.videoHeight).clamp(0.3, 4.0);
        final faceTransformMatrix =
          _toDoubleList(result['transformMatrix'], expectedLength: 16);

      tracking.value = tracking.value.copyWith(
        faceRect: Rect.fromLTWH(left, top, width, height),
        headYaw: yaw,
        headRoll: roll,
        eyeMidY: eyeMidY,
        eyeMidX: eyeMidX,
        eyeDistanceNorm: eyeDistanceNorm,
        videoAspect: videoAspect,
        faceTransformMatrix: faceTransformMatrix,
      );
    } catch (_) {
      return;
    }
  }

  Future<Uint8List?> capture() async {
    if (_video == null || _stream == null) {
      return null;
    }
    final width = _video!.videoWidth;
    final height = _video!.videoHeight;
    if (width == 0 || height == 0) {
      return null;
    }

    final canvas = html.CanvasElement(width: width, height: height);
    final ctx = canvas.context2D;
    ctx.drawImage(_video!, 0, 0);
    final dataUrl = canvas.toDataUrl('image/jpeg', 0.9);
    final base64Data = dataUrl.split(',').last;
    return Uint8List.fromList(base64Decode(base64Data));
  }

  Widget buildView() {
    if (_viewType == null) {
      return const SizedBox.shrink();
    }
    return HtmlElementView(viewType: _viewType!);
  }

  List<double>? _toDoubleList(dynamic value, {required int expectedLength}) {
    if (value == null || value is! js.JsObject) {
      return null;
    }

    final out = <double>[];
    for (var i = 0; i < expectedLength; i++) {
      final item = value[i];
      if (item is! num) {
        return null;
      }
      out.add(item.toDouble());
    }
    return out;
  }
}
